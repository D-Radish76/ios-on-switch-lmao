# The U-Boot half of the D1 loader diagnostic, sourced from its slot in bl33.bin (tools/package/diag_sd_loader.py
# fills in the values). It is D0-R's script plus the normal script's exporter and SD reference steps, run in the normal
# order, each behind checks the normal script does not make (ERISTA_D1_DESIGN.md): the exporter's header words and data
# CRC32 in its slot, the two cache helpers' instructions, its CRC32 again at the run address and again right before it
# runs. The card is read once, sector 0 of mmc 1, with no retry, no other device and no write. Nothing here saves the
# environment, writes the card or starts a kernel. A failed step names its status, waits @DWELL@ s on the screen and
# resets: the exporter and the loader are not started past it. The loader gets the facts through the status block at
# @STATUS@ (boot/diag/sd_loader/sd_report.h), which holds the flags this script sets and the CRC32 values it measured.
setenv handoff_status WRONG_BOARD
if setenv stdout vidconsole && setenv stderr vidconsole; then
    echoe "D1 LOADER DIAGNOSTIC. NO KERNEL. NO BOOT"
else
    setenv handoff_status CONSOLE_FAILED
fi
base 0
if test "${handoff_status}" = "WRONG_BOARD"; then
@BOARD_GATE@
fi
if test "${handoff_status}" = "READY"; then
    echoe "BOARD OK"
fi

if test "${handoff_status}" = "READY"; then
    setenv handoff_fb_addr
    setenv handoff_fb_size_bytes
    setenv handoff_fb_width
    setenv handoff_fb_height
    setenv handoff_fb_format
    setenv handoff_fb_rot
    if test -n "${fdtcontroladdr}" && fdt addr ${fdtcontroladdr}; then
        if fdt get value handoff_fb_addr /framebuffer address && fdt get size handoff_fb_size_bytes /framebuffer size && fdt get value handoff_fb_width /framebuffer width && fdt get value handoff_fb_height /framebuffer height && fdt get value handoff_fb_format /framebuffer format && fdt get value handoff_fb_rot /framebuffer rot; then
            if itest.l ${handoff_fb_addr} == 0xF5A00000 && itest.l ${handoff_fb_size_bytes} == 4 && itest.l ${handoff_fb_width} == 0x2D0 && itest.l ${handoff_fb_height} == 0x500 && test "${handoff_fb_format}" = "a8b8g8r8" && itest.l ${handoff_fb_rot} == 3; then
                echoe "FRAMEBUFFER OK"
            else
                setenv handoff_status FRAMEBUFFER_MISMATCH
            fi
        else
            setenv handoff_status FRAMEBUFFER_PROPERTY_FAILED
        fi
    else
        setenv handoff_status CONTROL_FDT_FAILED
    fi
fi

if test "${handoff_status}" = "READY"; then
    if cp.b @DTB@ 0x8D000000 @DTB_BYTES@; then
        if fdt addr 0x8D000000 0x30000; then
            if itest.l *0x8D000000 == 0xedfe0dd0 && itest.l *0x8D000004 == 0x00000300; then
                if fdt rsvmem add 0xF5A00000 0x400000; then
                    echoe "DTB COPIED AND FRAMEBUFFER RESERVED"
                else
                    setenv handoff_status FRAMEBUFFER_RESERVE_FAILED
                fi
            else
                setenv handoff_status DTB_PADDING_FAILED
            fi
        else
            setenv handoff_status DTB_HEADER_FAILED
        fi
    else
        setenv handoff_status DTB_COPY_FAILED
    fi
fi

if test "${handoff_status}" = "READY"; then
    if mw.b @STATUS@ 0 @STATUS_BYTES@; then
        mw.l @S_VERSION@ @STATUS_VERSION@ 1
        mw.l @S_EXPECTED_CRC@ @EXPECTED_CRC_STORED@ 1
        mw.l @S_PAYLOAD_BYTES@ @EXPORTER_BYTES@ 1
    else
        setenv handoff_status STATUS_BLOCK_FAILED
    fi
fi

# The exporter's own checks name devnum 1, so a script that reads another device stops here, before anything is loaded.
if test "${handoff_status}" = "READY"; then
    if test "${devnum}" != "1"; then
        setenv handoff_status SD_REFERENCE_WRONG_DEVICE
    fi
fi

# The image in its slot: all 16 header words, then its data CRC32. crc32 stores the CRC big endian, so the constant is the
# word itest.l reads back.
if test "${handoff_status}" = "READY"; then
    if @EXPORTER_HEADER_TESTS@; then
        mw.l @S_HEADER@ 1 1
    else
        setenv handoff_status EXPORTER_SLOT_HEADER_MISMATCH
    fi
fi
if test "${handoff_status}" = "READY"; then
    if crc32 @EXPORTER_DATA@ @EXPORTER_BYTES@ @S_CRC_SLOT@ && itest.l *@S_CRC_SLOT@ == @EXPECTED_CRC_STORED@; then
        mw.l @S_SLOT_CRC@ 1 1
    else
        setenv handoff_status EXPORTER_SLOT_CRC_MISMATCH
    fi
fi

# The two U-Boot routines the normal script runs to make the exporter visible to instruction fetch: the words, as the
# pinned bl33.bin holds them, must be in RAM where the script will call them.
if test "${handoff_status}" = "READY"; then
    if @HELPER_TESTS_0@ && @HELPER_TESTS_1@; then
        mw.l @S_HELPERS@ 1 1
    else
        setenv handoff_status CACHE_HELPER_MISMATCH
    fi
fi

# verify yes makes bootm start check the data CRC32 of the image in its slot (the pinned U-Boot skips it only for no);
# the loader's bootm below runs with verify no, as in D0.
if test "${handoff_status}" = "READY"; then
    if setenv verify yes && setenv autostart no && bootm start @EXPORTER@; then
        if bootm loados; then
            mw.l @S_LOADED@ 1 1
        else
            setenv handoff_status EXPORTER_LOAD_FAILED
        fi
    else
        setenv handoff_status EXPORTER_HEADER_FAILED
    fi
fi
setenv verify no

if test "${handoff_status}" = "READY"; then
    if crc32 @EXPORTER_RUN@ @EXPORTER_BYTES@ @S_CRC_LOADED@ && itest.l *@S_CRC_LOADED@ == @EXPECTED_CRC_STORED@; then
        mw.l @S_LOADED_CRC@ 1 1
        echoe "SD EXPORTER LOADED AND CHECKED"
    else
        setenv handoff_status EXPORTER_IMAGE_CORRUPT
    fi
fi

# Each helper returns 1 whatever it is given (it does not touch x0, which holds argc minus 1), so go reports failure;
# a go that reports success did not run the instructions the words above say.
if test "${handoff_status}" = "READY"; then
    if go @HELPER_0@; then
        setenv handoff_status CACHE_HELPER_RETURN_UNEXPECTED
    else
        if go @HELPER_1@; then
            setenv handoff_status CACHE_HELPER_RETURN_UNEXPECTED
        fi
    fi
fi


if test "${handoff_status}" = "READY"; then
    if setenv fdt_high ffffffffffffffff && setenv bootm_low 0x80000000 && setenv bootm_size 0x7FD00000 && setenv bootm_mapsize 0x7FD00000 && setenv bootargs switch_diag=1 && setenv verify no; then
        if bootm start @LOADER@ - 0x8D000000; then
            if bootm loados; then
                echoe "LOADER WRITTEN TO 0xA0000000"
            else
                setenv handoff_status IMAGE_LOAD_FAILED
            fi
        else
            setenv handoff_status BOOT_HEADER_FAILED
        fi
    else
        setenv handoff_status BOOT_ENVIRONMENT_FAILED
    fi
fi

if test "${handoff_status}" = "READY"; then
    if bootm prep; then
        echoe "READ SD SECTOR 0"
    else
        setenv handoff_status BOOT_PREP_FAILED
    fi
fi

# One attempt at one block of mmc 1, and nothing else touches the card: no mmc dev, no second device, no retry.
if test "${handoff_status}" = "READY"; then
    if mw.b 0xA001F000 0 0x1000; then
        if read mmc 1 0xA001F200 0 1; then
            mw.l @S_READ@ 1 1
        else
            setenv handoff_status SD_REFERENCE_READ_FAILED
        fi
    else
        setenv handoff_status SD_REFERENCE_CLEAR_FAILED
    fi
fi

# Right before the exporter runs: the bytes at the run address are the ones that were checked.
if test "${handoff_status}" = "READY"; then
    if crc32 @EXPORTER_RUN@ @EXPORTER_BYTES@ @S_CRC_BEFORE_GO@ && itest.l *@S_CRC_BEFORE_GO@ == @EXPECTED_CRC_STORED@; then
        mw.l @S_BEFORE_GO_CRC@ 1 1
    else
        setenv handoff_status EXPORTER_IMAGE_CHANGED
    fi
fi
if test "${handoff_status}" = "READY"; then
    if go @EXPORTER_RUN@; then
        mw.l @S_GO_RC0@ 1 1
        if itest.l *0xA001F400 == 0x4d435349 && itest.l *0xA001F404 == 1 && itest.l *0xA001F408 == 0x100; then
            mw.l @S_RECORD@ 1 1
        else
            setenv handoff_status SD_CARD_RECORD_FAILED
        fi
    else
        setenv handoff_status SD_CARD_EXPORT_FAILED
    fi
fi
if test "${handoff_status}" = "READY"; then
    if crc32 0xA001F200 0x200 0xA001F028 && mw.l 0xA001F004 2 1 && mw.l 0xA001F008 0x500 1 && mw.l 0xA001F00C 0x200 1 && mw.l 0xA001F010 0x200 1 && mw.l 0xA001F014 0x700B0000 1 && mw.l 0xA001F018 0 1 && mw.l 0xA001F01C 1 1 && mw.l 0xA001F020 1 1 && mw.l 0xA001F024 0 1 && mw.l 0xA001F02C 0 1 && mw.l 0xA001F000 0x52445349 1; then
        mw.l @S_REFERENCE@ 1 1
        mw.l @S_MAGIC@ @STATUS_MAGIC@ 1
        echoe "HANDOFF TO THE LOADER. THE SCREEN CLEARS, THEN THE REPORT"
        bootm go
        setenv handoff_status UNEXPECTED_BOOTM_RETURN
    else
        setenv handoff_status SD_REFERENCE_METADATA_FAILED
    fi
fi

echoe "REFUSED: ${handoff_status}. Reset in @DWELL@ s, expected to land in hekate, not verified"
for left in @DWELL_STEPS@; do
    sleep @STEP@
    echoe "RESET in ${left} s"
done
reset
while true; do
    sleep 1
done
