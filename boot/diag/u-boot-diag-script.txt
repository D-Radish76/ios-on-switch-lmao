# Read only memory controller diagnostic, sourced from its slot in this build's bl33.bin (tools/package/diag_boot.py).
# It reads the registers native_gate (boot/loader/loader.c) compares, prints them on the screen and stops. It carries no
# loader, kernel or exporter. Its only exit is the reset after two minutes, which ends in hekate, never in a kernel. diag_boot_test.py lists the commands it may use.
setenv diag_status WRONG_BOARD
setenv diag_reads 0
if setenv stdout vidconsole; then
    setenv diag_console OK
else
    setenv diag_console FAILED
fi

if test "${diag_console}" = "OK"; then
@BOARD_CHECK@
fi

echo DIAGNOSTIC: READ ONLY, NO BOOT
echo board t210b01=${t210b01} sku=${sku} status=${diag_status} console=${diag_console}
if test "${diag_status}" = "READY"; then
    echo each line is an address then its value. If it stops, the last line shown is the last register read
@READS@
    if test "${diag_reads}" = "@COUNT_HEX@"; then
        echo DONE: all @COUNT@ registers read
    else
        echo INCOMPLETE: ${diag_reads} of @COUNT_HEX@ read (hex), status=${diag_status}
    fi
else
    echo REFUSED: nothing was read, status=${diag_status} console=${diag_console}
fi
echo Photograph the screen now. It resets to hekate in 2 minutes. Or hold the power button.
for left in 90 60 30 0; do
    sleep 30
    echo resetting to hekate in ${left} s
done
reset
while true; do
    sleep 1
done
